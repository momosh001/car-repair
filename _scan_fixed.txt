function scan(file){
    if(busy) return;
    busy = true;
    showOverlay("Готовим фото…", "Держите VIN целиком в кадре");
    var canvas = null;
    btnCam.disabled = true;
    btnGal.disabled = true;

    function run(psm){
      return recognize(canvas, psm).then(function(data){
        var cands = findCandidates(data.text);
        return { list: cands.map(function(c){ return c.vin; }), conf: data.confidence || 0, all: cands };
      });
    }

    fileToCanvas(file).then(function(c){
      canvas = c;
      return run("7");
    }).then(function(a){
      if(a.list.length === 1 && a.all[0] && a.all[0].check) return { list: [a.list[0]], conf: a.conf, all: a.all };
      return run("6").then(function(b){
        var v = vote([a, b]);
        if(v.list.length && v.all[0] && v.all[0].check) return { list: [v.list[0]], conf: v.conf, all: v.all };
        return run("11").then(function(c){ return vote([a, b, c]); });
      });
    }).then(function(res){
      hideOverlay();
      showResult(res.list, res.conf, res.all || []);
    }).catch(function(err){
      hideOverlay();
      showResult([], 0, [], (err && err.message) || "");
    }).then(function(){
      busy = false;
      btnCam.disabled = false;
      btnGal.disabled = false;
    });
  }
