function showResult(list, conf, all, errMsg){
    all = all || [];
    var ok = !!(list && list.length);
    current = ok ? list[0] : "";
    $("#scanTitle").textContent = ok ? "VIN распознан" : "Не удалось распознать";
    $("#scanVin").textContent = ok ? current : "—";
    $("#scanApply").style.display = ok ? "" : "none";
    $("#scanRetryRow").style.display = ok ? "none" : "flex";

    var candBox = $("#scanCandidates");
    if(ok && all && all.length > 1){
      candBox.style.display = "";
      candBox.innerHTML = all.slice(0,4).map(function(c, idx){
        var mark = c.check ? " <span style='color:var(--accent);font-weight:700'>✓</span>" : "";
        var checked = (idx === 0) ? 'checked' : '';
        return '<label style="display:block;padding:6px 10px;border:1px solid var(--line);border-radius:8px;margin:6px 0;cursor:pointer"><input type="radio" name="cand" value="'+c.vin+'" style="margin-right:8px" onclick="current=this.value" '+checked+'>'+c.vin+mark+'</label>';
      }).join("");
    }else{
      candBox.style.display = "none";
      candBox.innerHTML = "";
    }

    var checkText = (all&&all[0]&&all[0].check)?"совпадает":"не подтверждена";
    $("#scanHint").textContent = ok
      ? "Проверьте символы: часто путаются 0/О, 1/I, 8/B, 5/S. Контрольная цифра VIN: "+checkText+". Если не совпало — исправьте вручную."
      : (errMsg || "Сфотографируйте табличку VIN крупно и ровно, без бликов, целиком в кадре. Или введите VIN вручную.");
    $("#scanModal").classList.add("on");
  }
